\section*{Keywords}
\begin{tabularx}{\textwidth}{ l X }
    \textit{door-in-the-face} & a compliance technique where an initial large request is made, which is expected to be refused, followed by a smaller, more reasonable request \\
    \textit{foot-in-the-door} & a compliance technique where a small initial request is made to gain agreement, followed by a larger request \\
    \textit{low-balling technique} & a persuasion tactic in which an item is initially offered at a lower price than one expects in order to get the buyer to commit; then the price is suddenly increased \\ 
    \textit{endless-chain} & a sales technique in which a salesperson attempts to persuade a customer to recommend friends or family members who might also make a purchase \\
\end{tabularx}
